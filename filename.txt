import numpy as np

class QLearningAgent:
    def __init__(
        self,
        state_size,
        action_size,
        learning_rate=0.8,
        discount_factor=0.95
    ):
        self.q_table = np.zeros((state_size, action_size))
        self.learning_rate = learning_rate
        self.discount_factor = discount_factor

    def choose_action(self, state, epsilon, env):
        if np.random.random() < epsilon:
            return env.action_space.sample()
        return int(np.argmax(self.q_table[state]))

    def learn(self, state, action, reward, next_state, terminated):
        next_value = np.max(self.q_table[next_state])
        target = reward + (
            self.discount_factor * next_value * (not terminated)
        )
        self.q_table[state, action] += self.learning_rate * (
            target - self.q_table[state, action]
        )
